country_event = {
    id = ark_gateway.1
    hide_window = yes
    is_triggered_only = yes
    immediate = {
        create_country = {
            name = "NAME_ark_gateway"
            type = ark_country_type_gateway
            flag = {
                icon = {
                    category = "legion"
                    file = "flag_mp_beta_05.dds"
                }
                background = {
                    category = "backgrounds"
                    file = "00_solid.dds"
                }
                colors = {
                    "dark_blue"
                    "dark_blue"
                    "null"
                    "null"
                }
            }
            effect = {
                save_global_event_target_as = ark_gateway_country
                set_country_flag = ark_gateway_country
            }
        }
        event_target:ark_gateway_archaeology_location = {
            create_fleet = {
                name = "NAME_ark_gateway_transporter"
                settings = {
                    can_disband = no
                }
                effect = {
                    set_owner = event_target:ark_gateway_country
                    set_fleet_flag = ark_gateway_main_fleet
                    create_ship = {
                        name = "NAME_ark_gateway_transporter"
                        design = "NAME_Monadic_Qerr"
                        effect = {
                            add_modifier = {
                                modifier = ark_ship_no_damage_received_modifier
                                days = -1
                            }
                        }
                    }
                    set_location = prev
                }
            }
            destroy_ambient_object = this
        }
        last_created_fleet = {
            if = {
                limit = {
                    event_target:ark_country = {has_country_flag = ark_gateway_arrive_no_hyperlane}
                }
                set_fleet_flag = ark_gateway_main_fleet_through_space
                event_target:ark_gateway_archaeology_system = {
                    create_ambient_object = {
                        type = "ark_ambient_null"
                        location = this.star
                    }
                    last_created_ambient_object = {
                        set_location = {
                            target = prev
                            distance = 600
                            angle = random
                        }
                        save_event_target_as = ark_gateway_through_space_target
                    }
                }

                auto_move_to_planet = {
                    target = event_target:ark_gateway_through_space_target
                    clear_auto_move_on_arrival = no
                }
                set_variable = {
                    which = var_ark_gateway_voyage_ship_accelerate_timer
                    value = 0
                }
                fleet_event = { id = ark_gateway.3231 }
            }
            if = {
                limit = {
                    NOT = {
                        event_target:ark_country = {has_country_flag = ark_gateway_arrive_no_hyperlane}
                    }
                }
                auto_move_to_planet = {
                    target = event_target:ARK_TYRA_HOME_SYSTEM.star
                    clear_auto_move_on_arrival = no
                    destroy_on_arrival = yes
                }
            }
        }
    }
}

country_event = {
    id = ark_gateway.11
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        OR = {
            AND = {
                from = { has_fleet_flag = ark_gateway_main_fleet }
                solar_system = { NOT = { has_star_flag = ark_gateway_constructing } }
            }
            AND = { has_country_flag = ark_gateway_manually_triggered }
        }
    }

    immediate = {

        solar_system = {
            star = {
                save_global_event_target_as = ark_gateway_source_star
            }
            set_star_flag = ark_gateway_constructing
            spawn_megastructure = {
                type = mega_shipyard_0
                planet = star
                init_effect = {
                    set_owner = event_target:ark_country
                }
            }
            export_trigger_value_to_variable = {
                trigger = count_neighbor_system
                variable = var_ark_neighbor_systems
            }

            while = {
                count = var_ark_neighbor_systems

                random_neighbor_system = {
                    limit = {
                        NOT = { has_star_flag = ark_gateway_target_end }
                    }
                    star = {
                        save_event_target_as = ark_gateway_target_star
                    }
                }

                create_fleet = {
                    name = "NAME_ark_gateway_prospector"
                    settings = {
                        can_disband = no
                    }
                    effect = {
                        set_owner = event_target:ark_country
                        create_ship = {
                            name = "NAME_ark_gateway_prospector"
                            design = "NAME_Monadic_Qerr"
                            effect = {
                                add_modifier = {
                                    modifier = ark_ship_no_damage_received_modifier
                                    days = -1
                                }
                            }
                        }
                        set_location = prev
                    }
                }

                last_created_fleet = {
                    auto_move_to_planet = {
                        target = event_target:ark_gateway_target_star
                        clear_auto_move_on_arrival = yes
                        #arrival_effect = ark_gateway_arrival_effect_back
                    }
                    set_fleet_flag = ark_gateway_sub_fleet
                }
                event_target:ark_gateway_target_star = {
                    solar_system = { set_star_flag = ark_gateway_target_end }
                }
                log = "ARK 探针已出发"
            }
            log = "ARK 路网已锚定"
            #every_neighbor_system = {
            #    remove_star_flag = ark_gateway_target_end
            #}
            if = {
                limit = { event_target:ark_country = { NOT = { has_country_flag = ark_gateway_communicate_1 } } }
                event_target:ark_country = {
                    country_event = {
                        id = ark_gateway.111
                    }
                    set_country_flag = ark_gateway_communicate_1
                }
            }
        }
    }
}

fleet_event = {
    id = ark_gateway.13
    is_triggered_only = yes
    hide_window = yes
    trigger = {
        has_fleet_flag = ark_gateway_sub_fleet
        from = {
            has_star_flag = ark_gateway_constructing
        }
    }
    immediate = {
        log = "ARK 探针已返航"
        destroy_fleet = this
        create_ambient_object = {
            type = ark_arclight_one_object
            location = this
            scale = 1.0
            use_3d_location = yes
            entity_offset = 0
            entity_offset_angle = 0
            entity_offset_height = 0
            base_angle_towards = star
            #entity_face_object =
            #entity_scale_to_size =
            #scripted_scale =
            #play_animation_once =
            #duration =
            #effect =
        }
        if = {
            limit = { event_target:ark_country = { NOT = { has_country_flag = ark_gateway_communicate_3 } } }
            event_target:ark_country = {
                country_event = {
                    id = ark_gateway.131
                }
                set_country_flag = ark_gateway_communicate_3
            }
        }
        from = {
            change_variable = {
                which = var_ark_neighbor_systems
                value = -1
            }
            if = {
                limit = {
                    check_variable = {
                        which = var_ark_neighbor_systems
                        value = 0
                    }
                }
                if = {
                    limit = { event_target:ark_country = { NOT = { has_country_flag = ark_gateway_communicate_4 } } }
                    event_target:ark_country = {
                        country_event = {
                            id = ark_gateway.132
                        }
                        set_country_flag = ark_gateway_communicate_4
                    }
                }
                every_neighbor_system = {
                    remove_star_flag = ark_gateway_sub_system
                    remove_star_flag = ark_gateway_target_end
                    remove_star_flag = ark_gateway_constructing
                }
                remove_star_flag = ark_gateway_sub_system
                remove_star_flag = ark_gateway_target_end
                remove_star_flag = ark_gateway_constructing
                clear_global_event_target = ark_gateway_source_star
                clear_variable = var_ark_neighbor_systems
                set_star_flag = ark_gateway_constructed
                log = "ARK 路网展开完毕"
            }
        }
    }
}
